	name = TNO_RegionalOrganization_GetMapmodeTooltip
	text = { # Tag is part of the organization

	}

}

# Used to populate the effects tooltip in applicable regorgs
defined_text = {
	name = TNO_RegionalOrganization_GetModifiers
	text = {
		trigger = {
			NOT = { check_variable = { modifiers@var:reg_org^num = 0 } }
			set_temp_variable = { Recursion_i = -1 }
		}
		localization_key = "RegionalOrganization_Icon_tt_modifiers"
	}
}
defined_text = {
	name = TNO_RegionalOrganization_GetModifiersEntry
	text = {
		trigger = {
			add_to_temp_variable = { Recursion_i = 1 } # Continues the loop
			check_variable = { modifiers@var:reg_org^num > Recursion_i } # Used to terminate the loop
			NOT = { check_variable = { modifiers@var:reg_org^Recursion_i = 0 } } # Skip over unset modifiers
		}
		localization_key = "[?reg_org.GetTokenKey]_modifier_[?Recursion_i]"
	}
	text = {
		trigger = { check_variable = { modifiers@var:reg_org^num > Recursion_i } } # If the above conditions return false for but the loop still hasn't been terminated, it will simply just skip it and continue
		localization_key = "[TNO_RegionalOrganization_GetModifiersEntry]"
	}
}
