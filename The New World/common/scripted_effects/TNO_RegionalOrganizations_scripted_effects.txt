### INFORMATION & DOCUMENTATION ###
# Regional Orgs have four types of information, encoded towards the org's idea token:
# 	RegionalOrganization_Leader - Leader of the regional org (tag)
# 	RegionalOrganization_Alignment - International alignment of the regional org (tag, set 0 for unaligned)
# 	RegionalOrganization_Members - Array list of all members
# 	RegionalOrganization_Desc - Description of the regional org. Default is 0, it's not something you need to specify unless you're implementing an alternate desc
# These three are stored in the global scope. There are two additional org list arrays:
#	global.TNO_RegionalOrganizations is a list of all active Regional Orgs on the globe
#	TNO_RegionalOrganizations (TAG-SCOPED, NON-GLOBAL) is a list of all orgs a givev tag is part of
# There is a given limit of how many orgs a given tag can be part of, this is defined by the @MaxOrgLimit constant
# Alignment is defined towards either one of the three superpowers (USA, GER, JAP), if the var is set at 0 its considered unaligned

### VALUE DEFINES ###
@MaxOrgLimit = 4 # Maximum amount of Regional Orgs that a tag can be into at once

## Adds a country to a Regional Organization
# Arguments: reg_org
RegionalOrganizations_Add = {
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Add; Executing"
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Add; Given: reg_org: [?reg_org.GetTokenKey]"

	if = {
		limit = { NOT = { has_variable = reg_org } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_Add; Handed unset org. Breaking."
	}
	else_if = {
		limit = { check_variable = { TNO_RegionalOrganizations^num = @MaxOrgLimit } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_Add; Tag already hit org limit. Breaking."
	}
	else = {
		custom_effect_tooltip = TNO_RegionalOrganizations_Add_tt
		add_to_array = { TNO_RegionalOrganizations = reg_org } # In-tag orgs list
		add_to_array = { global.RegionalOrganization_Members@var:reg_org = THIS }
		if = { # For econ-oriented orgs
			limit = { has_variable = global.RegionalOrganization_TotalGDP@var:reg_org }
			add_to_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = THIS.GDP }
		}
		hidden_effect = {
			meta_effect = {
				text = { add_ideas = [REG_ORG] }
				REG_ORG = "[?reg_org.GetTokenKey]"
			}

			# Joining a regorg increases trade influence with the leader
			var:global.RegionalOrganization_Leader@var:reg_org = {
				set_temp_variable = { temp_modifier = 20 }
				add_to_trade_modifier_PREV = yes 
			}
		}
		force_update_map_mode = { mapmode = tno_regional_organizations_map_mode }
	}

	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Add; Completed execution."
}

## Removes a country to a Regional Organization
# Arguments: reg_org
RegionalOrganizations_Remove = {
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Remove; Executing"
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Remove; Given: reg_org: [?reg_org.GetTokenKey]"

	if = {
		limit = { NOT = { has_variable = reg_org } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_Remove; Handed unset org. Breaking."
	}
	else = {
		custom_effect_tooltip = TNO_RegionalOrganizations_Remove_tt
		remove_from_array = { TNO_RegionalOrganizations = reg_org }
		remove_from_array = { global.RegionalOrganization_Members@var:reg_org = THIS }
		if = { # For econ-oriented orgs
			limit = { has_variable = global.RegionalOrganization_TotalGDP@var:reg_org }
			subtract_from_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = THIS.GDP }
		}
		hidden_effect = {
			meta_effect = {
				text = { remove_ideas = [REG_ORG] }
				REG_ORG = "[?reg_org.GetTokenKey]"
			}

			var:global.RegionalOrganization_Leader@var:reg_org = {
				set_temp_variable = { temp_modifier = -20 }
				add_to_trade_modifier_PREV = yes 
			}
		}
		force_update_map_mode = { mapmode = tno_regional_organizations_map_mode }

		if = { # Disband a Regional Organization if the last country in it is being removed
			limit = { check_variable = { global.RegionalOrganization_Members@var:reg_org^num = 0 } }
			RegionalOrganizations_Disband = yes
		}
	}

	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Remove; Completed execution."
}

## Initializes a new Regional Organization. Takes reg_org as the argument for the new org. Sets the tag this has been scoped in as the org leader by default
# Arguments: reg_org, org_alignment, org_leader (optional), org_type (optional)
RegionalOrganizations_Initialize = {
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Initialize; Executing"
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Initialize; Given: reg_org: [?reg_org.GetTokenKey], org_alignment: [?org_alignment.GetTag], org_leader: [?org_leader.GetTag], org_type: [?org_type]"

	if = {
		limit = { NOT = { has_variable = reg_org } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_Initialize; Handed unset org. Breaking."
	}
	else = {
		custom_effect_tooltip = TNO_RegionalOrganizations_Initialize_tt
		add_to_array = { global.TNO_RegionalOrganizations = reg_org } # Global orgs list
		set_variable = { global.RegionalOrganization_Leader@var:reg_org = org_leader?THIS }
		set_variable = { global.RegionalOrganization_Alignment@var:reg_org = org_alignment?THIS }
		set_variable = { global.RegionalOrganization_Type@var:reg_org = org_type }
		if = { # Optional setup for orgs in which their combined GDP is important, sets it up to be updated every month
			limit = { TNO_RegionalOrganizations_EconRelevantOrg = yes }
			set_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = 0 }
			add_to_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org }
		}
		hidden_effect = { RegionalOrganizations_Add = yes }
		var:global.RegionalOrganization_Leader@var:reg_org = { add_to_array = { TNO_RegionalOrganizations_Led = reg_org } }
	}

	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Initialize; Completed execution."
}

## Disbands a Regional Organization, removing it from the list
# Arguments: reg_org
RegionalOrganizations_Disband = {
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Disband; Executing"
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Disband; Given: reg_org: [?reg_org.GetTokenKey]"

	if = {
		limit = { NOT = { has_variable = reg_org } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_Disband; Handed unset org. Breaking."
	}
	else = {
		custom_effect_tooltip = TNO_RegionalOrganizations_Disband_tt
		remove_from_array = { global.TNO_RegionalOrganizations = reg_org }
		clear_variable = global.RegionalOrganization_Leader@var:reg_org
		clear_variable = global.RegionalOrganization_Alignment@var:reg_org
		if = { # For econ-oriented orgs
			limit = { has_variable = global.RegionalOrganization_TotalGDP@var:reg_org }
			clear_variable = global.RegionalOrganization_TotalGDP@var:reg_org
			remove_from_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org }
		}

		if = { # Clear all members
			limit = { check_variable = { global.RegionalOrganization_Members@var:reg_org^num > 0 } }
			hidden_effect = {
				for_each_scope_loop = {
					array = global.RegionalOrganization_Members@var:reg_org
					remove_from_array = { TNO_RegionalOrganizations = reg_org }
					meta_effect = { text = { remove_ideas = [REG_ORG] } REG_ORG = "[?reg_org.GetTokenKey]" }
					var:global.RegionalOrganization_Leader@var:reg_org = {
						set_temp_variable = { temp_modifier = -20 }
						add_to_trade_modifier_PREV = yes 
					}
				}
			}
			clear_array = global.RegionalOrganization_Members@var:reg_org
			force_update_map_mode = { mapmode = tno_regional_organizations_map_mode }
		}

		var:global.RegionalOrganization_Leader@var:reg_org = { remove_from_array = { TNO_RegionalOrganizations_Led = reg_org } }
	}

	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_Disband; Completed execution."
}

## On_Monthly pulse
# Refreshes GDP values of every relevant org
RegionalOrganizations_MonthlyPulse = {
	for_each_loop = {
		array = global.TNO_RegionalOrganizations_EconOrgs
		set_temp_variable = { reg_org = v }
		RegionalOrganizations_TotalGDPRefresh = yes
	}
}

## Total GDP recalculation
RegionalOrganizations_TotalGDPRefresh = {
	set_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = 0 }
	for_each_scope_loop = {
		array = global.TNO_RegionalOrganizations_Members@var:reg_org
		add_to_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = THIS.GDP }
	}
}

RegionalOrganizations_ChangeType = {
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_ChangeType; Executing"
	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_ChangeType; Given: reg_org: [?reg_org.GetTokenKey], org_alignment: [?org_alignment.GetTag], org_type: [?org_type]"

	if = {
		limit = { NOT = { has_variable = reg_org } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_ChangeType; Handed unset org. Breaking."
	}
	else_if = {
		limit = { NOT = { has_variable = org_type } }
		log = "(ERROR): [GetDateText]: [THIS.GetName]: RegionalOrganizations_ChangeType; Handed unset leader. Breaking."
	}
	else = {
		custom_effect_tooltip = TNO_RegionalOrganizations_ChangeType_tt

		if = { # If it is now an econ-relevant org when it wasn't one before, add it to the array
			limit = {
				TNO_RegionalOrganizations_EconRelevantOrg = yes
				NOT = { is_in_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org } }
			}
			set_variable = { global.RegionalOrganization_TotalGDP@var:reg_org = 0 }
			add_to_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org }
			RegionalOrganizations_TotalGDPRefresh = yes
		}
		else_if = { # If its the inverse, remove it from the array
			limit = {
				TNO_RegionalOrganizations_EconRelevantOrg = no
				is_in_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org }
			}
			clear_variable = global.RegionalOrganization_TotalGDP@var:reg_org
			remove_from_array = { global.TNO_RegionalOrganizations_EconOrgs = reg_org }
		}

		set_variable = { global.RegionalOrganization_Type@var:reg_org = org_type }
	}

	log = "[GetDateText]: [THIS.GetName]: RegionalOrganizations_ChangeType; Completed execution."
}

# Scripted effect for Regional Organizations map mode
regional_organizations_mapmode_get_org_members = {
	if = {
		limit = {
			check_variable = {
				var = temp_org_idx
				value = global.RegionalOrganization_Members@var:reg_org^num
				compare = less_than
			}
		}
		if = {
			limit = {
				check_variable = { global.RegionalOrganization_Members@var:reg_org^temp_org_idx = FROM }
			}

			# Nothing happens and the loop is continued. This is so that sphere leader isn't displayed in list.
			add_to_temp_variable = { temp_org_idx = 1 }
			regional_organizations_mapmode_get_org_members = yes

			else = {
				custom_effect_tooltip = regional_organizations_map_mode_mapicon_tooltip_entry
				add_to_temp_variable = { temp_org_idx = 1 }
				regional_organizations_mapmode_get_org_members = yes
			}
		}
	}
}
